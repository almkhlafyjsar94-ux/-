name: Build JASSAR Games APK

on:
  workflow_dispatch:
  push:
    paths:
      - 'project.zip'
      - '.github/workflows/build-apk.yml'

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Get project files
        uses: actions/checkout@v4

      - name: Unpack Android project
        run: |
          mkdir source
          unzip -q project.zip -d source

      - name: Set up Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Android SDK
        uses: android-actions/setup-android@v4
        with:
          packages: 'platform-tools platforms;android-35 build-tools;35.0.0'

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Build APK
        run: gradle -p source :app:assembleDebug

      - name: Save APK
        uses: actions/upload-artifact@v4
        with:
          name: JASSAR-GAMES-APK
          path: source/app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
